<script>
function toggleMute() {
  const audio = document.getElementById('bg-music');
  const btn = document.querySelector('.mute-btn');
  if (audio.muted) {
    audio.muted = false;
    btn.textContent = '🔈 音乐';
  } else {
    audio.muted = true;
    btn.textContent = '🔇 静音';
  }
}
</script>
